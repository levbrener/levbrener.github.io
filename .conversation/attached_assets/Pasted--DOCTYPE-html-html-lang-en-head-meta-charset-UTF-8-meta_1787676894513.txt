  // -------- FRONT PAGE (with infinite scroll) --------
  let allFilteredStories = [];
  let loadedCount = 0;
  const BATCH_SIZE = 10;
  let isLoading = false;

  function renderFrontPage(){
    const container = document.getElementById('frontPageView');
    let pool = getVisibleStories();

    const archiveVal = document.getElementById('archiveFilter').value;
    if(archiveVal !== 'all'){
      const [year, month] = archiveVal.split('-').map(Number);
      pool = pool.filter(s => {
        const d = new Date(s.timestamp);
        return d.getFullYear() === year && d.getMonth() === month;
      });
    }

    const searchTerm = document.getElementById('searchInput').value.trim().toLowerCase();
    if(searchTerm){
      pool = pool.filter(s => s.headline.toLowerCase().includes(searchTerm) || (s.body && s.body.toLowerCase().includes(searchTerm)) || s.section.toLowerCase().includes(searchTerm) || (s.byline && s.byline.toLowerCase().includes(searchTerm)) || (s.tags && s.tags.toLowerCase().includes(searchTerm)) || (s.dek && s.dek.toLowerCase().includes(searchTerm)));
    }
    if(currentFilter !== 'all') pool = pool.filter(s => s.section === currentFilter);
    if(showSavedOnly) pool = pool.filter(s => isSaved(s.id));

    allFilteredStories = pool;
    loadedCount = 0;
    container.innerHTML = '';
    loadMoreBatch();
  }

  function loadMoreBatch() {
    const container = document.getElementById('frontPageView');
    if (!container) return;
    if (isLoading) return;
    const remaining = allFilteredStories.slice(loadedCount);
    if (remaining.length === 0) {
      const sentinel = document.getElementById('loadSentinel');
      if (sentinel) sentinel.remove();
      return;
    }
    isLoading = true;
    const batch = remaining.slice(0, BATCH_SIZE);
    loadedCount += batch.length;

    const trending = [...allFilteredStories].sort((a,b) => (b.viewCount||0) - (a.viewCount||0)).slice(0,3);
    const searchTerm = document.getElementById('searchInput').value.trim();

    let html = '';
    const isFirstBatch = (loadedCount === batch.length);
    if (isFirstBatch) {
      const lead = batch[0] || allFilteredStories[0];
      const usedIds = new Set([lead.id]);
      const sideStories = batch.filter(s => s.id !== lead.id).slice(0,3);
      sideStories.forEach(s => usedIds.add(s.id));

      html += '<section class="lead-grid"><div class="lead-story">' +
        '<span class="kicker">' + escapeHtml(lead.section) + (trending.some(t => t.id === lead.id) ? ' <span class="trending-badge">🔥 Trending</span>' : '') + (lead.isBreaking ? ' <span class="trending-badge" style="background:#c0392b;">🚨 Breaking</span>' : '') + '</span>' +
        '<h2><a data-goto="' + lead.id + '">' + highlightText(escapeHtml(lead.headline), searchTerm) + '</a></h2>' +
        '<img src="' + (lead.imageUrl || placeholderFor(lead.id)) + '" alt="' + escapeHtml(lead.imageCaption || lead.headline) + '" loading="lazy" onclick="window.openLightbox(\'' + lead.id + '\', 0)">' +
        (lead.imageCaption ? '<p class="caption">' + escapeHtml(lead.imageCaption) + '</p>' : '<div style="margin-bottom:18px;"></div>') +
        (lead.dek ? '<p class="dek">' + highlightText(escapeHtml(lead.dek), searchTerm) + '</p>' : '') +
        '<span class="byline">' + (lead.byline ? 'By ' + escapeHtml(lead.byline) + ' · ' : '') + timeAgo(lead.timestamp) + ' · 👁 ' + (lead.viewCount||0) + ' · ' + readingTime(lead) + ' · <span class="save-star" data-id="' + lead.id + '" style="cursor:pointer;">' + (isSaved(lead.id) ? '⭐' : '☆') + '</span></span>' +
        '</div><div class="sidebar">' +
        sideStories.map(s =>
          '<div class="side-story"><span class="kicker">' + escapeHtml(s.section) + (trending.some(t => t.id === s.id) ? ' <span class="trending-badge">🔥</span>' : '') + (s.isBreaking ? ' <span class="trending-badge" style="background:#c0392b;">🚨</span>' : '') + '</span>' +
          '<h3><a data-goto="' + s.id + '">' + highlightText(escapeHtml(s.headline), searchTerm) + '</a></h3>' +
          '<p>' + highlightText(escapeHtml(s.dek || ''), searchTerm) + '</p>' +
          '<span class="byline"><span class="save-star" data-id="' + s.id + '" style="cursor:pointer;">' + (isSaved(s.id) ? '⭐' : '☆') + '</span></span></div>'
        ).join('') +
        '</div></section>';

      const remainingCards = batch.filter(s => !usedIds.has(s.id));
      if (remainingCards.length) {
        html += '<div class="story-row">' + remainingCards.map(s => cardHtml(s, trending, searchTerm)).join('') + '</div>';
      }
    } else {
      html += '<div class="story-row">' + batch.map(s => cardHtml(s, trending, searchTerm)).join('') + '</div>';
    }

    const oldSentinel = document.getElementById('loadSentinel');
    if (oldSentinel) oldSentinel.remove();

    container.insertAdjacentHTML('beforeend', html);

    container.querySelectorAll('[data-goto]').forEach(el => { el.style.cursor = 'pointer'; el.onclick = () => goToStory(el.dataset.goto); });
    container.querySelectorAll('.save-star').forEach(el => {
      el.onclick = (e) => { e.stopPropagation(); toggleSaved(el.dataset.id); };
    });
    container.querySelectorAll('.lead-story img, .card img').forEach(img => {
      img.style.cursor = 'pointer';
      img.onclick = function(e) {
        const storyId = this.closest('[data-story-id]')?.dataset.storyId || this.closest('.lead-story')?.querySelector('[data-goto]')?.dataset.goto;
        if (storyId) {
          const images = getAllImagesForStory(storyId);
          const idx = images.findIndex(i => i.src === this.src);
          openLightbox(storyId, idx > -1 ? idx : 0);
        }
      };
    });

    if (loadedCount < allFilteredStories.length) {
      const sentinel = document.createElement('div');
      sentinel.id = 'loadSentinel';
      sentinel.style.height = '20px';
      sentinel.style.textAlign = 'center';
      sentinel.style.padding = '20px';
      sentinel.textContent = 'Loading more…';
      container.appendChild(sentinel);
      const observer = new IntersectionObserver((entries) => {
        if (entries[0].isIntersecting) {
          observer.disconnect();
          loadMoreBatch();
        }
      });
      observer.observe(sentinel);
    }

    isLoading = false;
    updateSavedUI();
  }

  function cardHtml(s, trending, searchTerm) {
    return '<div class="card" data-story-id="' + s.id + '"><img src="' + (s.imageUrl || placeholderFor(s.id)) + '" alt="' + escapeHtml(s.imageCaption || s.headline) + '" loading="lazy">' +
      '<span class="kicker">' + escapeHtml(s.section) + (trending.some(t => t.id === s.id) ? ' <span class="trending-badge">🔥</span>' : '') + (s.isBreaking ? ' <span class="trending-badge" style="background:#c0392b;">🚨</span>' : '') + '</span>' +
      '<h3><a data-goto="' + s.id + '">' + highlightText(escapeHtml(s.headline), searchTerm) + '</a></h3>' +
      '<p>' + highlightText(escapeHtml(s.dek || ''), searchTerm) + '</p>' +
      '<span class="byline">' + (s.byline ? 'By ' + escapeHtml(s.byline) : '') + ' · ' + readingTime(s) + ' · <span class="save-star" data-id="' + s.id + '" style="cursor:pointer;">' + (isSaved(s.id) ? '⭐' : '☆') + '</span></span></div>';
  }

  function readingTime(story) {
    const words = (story.body || '').split(/\s+/).length;
    const mins = Math.max(1, Math.round(words / 200));
    return mins + ' min read';
  }

  function highlightText(text, term) {
    if (!term || !text) return text;
    const regex = new RegExp(term.replace(/[.*+?^${}()|[\]\\]/g,'\\$&'), 'gi');
    return text.replace(regex, '<mark>$&</mark>');
  }

  function getAllImagesForStory(storyId) {
    const story = stories.find(s => s.id === storyId);
    if (!story) return [];
    const images = [];
    if (story.imageUrl) images.push({ src: story.imageUrl, caption: story.imageCaption || '' });
    return images;
  }

  // -------- LIGHTBOX --------
  function openLightbox(storyId, index) {
    const images = getAllImagesForStory(storyId);
    if (!images.length) return;
    lightboxImages = images;
    lightboxIndex = Math.min(index, images.length - 1);
    const lb = document.getElementById('lightbox');
    const img = document.getElementById('lbImage');
    const cap = document.getElementById('lbCaption');
    img.src = lightboxImages[lightboxIndex].src;
    cap.textContent = lightboxImages[lightboxIndex].caption || '';
    lb.classList.add('active');
    document.body.style.overflow = 'hidden';
    updateLightboxNav();
  }

  function openLightboxWithImages(images, index) {
    if (!images || !images.length) return;
    lightboxImages = images;
    lightboxIndex = Math.min(index, images.length - 1);
    const lb = document.getElementById('lightbox');
    const img = document.getElementById('lbImage');
    const cap = document.getElementById('lbCaption');
    img.src = lightboxImages[lightboxIndex].src;
    cap.textContent = lightboxImages[lightboxIndex].caption || '';
    lb.classList.add('active');
    document.body.style.overflow = 'hidden';
    updateLightboxNav();
  }

  function closeLightbox() {
    document.getElementById('lightbox').classList.remove('active');
    document.body.style.overflow = '';
  }

  function lightboxPrev() {
    if (lightboxImages.length === 0) return;
    lightboxIndex = (lightboxIndex - 1 + lightboxImages.length) % lightboxImages.length;
    const img = document.getElementById('lbImage');
    const cap = document.getElementById('lbCaption');
    img.src = lightboxImages[lightboxIndex].src;
    cap.textContent = lightboxImages[lightboxIndex].caption || '';
    updateLightboxNav();
  }

  function lightboxNext() {
    if (lightboxImages.length === 0) return;
    lightboxIndex = (lightboxIndex + 1) % lightboxImages.length;
    const img = document.getElementById('lbImage');
    const cap = document.getElementById('lbCaption');
    img.src = lightboxImages[lightboxIndex].src;
    cap.textContent = lightboxImages[lightboxIndex].caption || '';
    updateLightboxNav();
  }

  function updateLightboxNav() {
    const prev = document.getElementById('lbPrev');
    const next = document.getElementById('lbNext');
    if (lightboxImages.length <= 1) {
      prev.style.display = 'none';
      next.style.display = 'none';
    } else {
      prev.style.display = 'block';
      next.style.display = 'block';
    }
  }

  window.openLightbox = openLightbox;
  window.openLightboxWithImages = openLightboxWithImages;

  document.getElementById('lbClose').addEventListener('click', closeLightbox);
  document.getElementById('lbPrev').addEventListener('click', lightboxPrev);
  document.getElementById('lbNext').addEventListener('click', lightboxNext);
  document.addEventListener('keydown', (e) => {
    if (document.getElementById('lightbox').classList.contains('active')) {
      if (e.key === 'Escape') closeLightbox();
      if (e.key === 'ArrowLeft') lightboxPrev();
      if (e.key === 'ArrowRight') lightboxNext();
    }
  });

  async function renderArticle(id){
    const container = document.getElementById('articleView');
    let s = stories.find(st => st.id === id);
    if(!s){
      container.innerHTML = '<div class="not-found"><p>Story not found.</p><p style="margin-top:14px;"><a class="back-link" id="notFoundBack">← Back</a></p></div>';
      document.getElementById('notFoundBack').onclick = goHome;
      return;
    }
    currentArticleId = id;
    if(!s.viewCount) s.viewCount = 0;
    s.viewCount += 1;
    await saveStoryToStorage(s);
    stories = stories.map(st => st.id === s.id ? s : st);

    // Open Graph
    const baseUrl = window.location.href.split('#')[0];
    document.querySelector('meta[property="og:title"]').content = s.headline + ' – The Wheaton Weekly';
    document.querySelector('meta[property="og:description"]').content = s.dek || s.headline;
    document.querySelector('meta[property="og:image"]').content = s.imageUrl || placeholderFor(s.id);
    document.querySelector('meta[property="og:url"]').content = baseUrl + '#/story/' + s.id;
    document.title = s.headline + ' – The Wheaton Weekly';

    // ---- Markdown paragraphs and headings ----
    const rawBody = s.body || '';
    const paragraphs = rawBody.split(/\n\s*\n/).filter(p => p.trim());
    const bodyHtml = paragraphs.map(p => '<p>' + markdownToHtml(p.trim()) + '</p>').join('');

    // Extract headings for TOC
    const headings = [];
    rawBody.split('\n').forEach(line => {
      if (line.trim().startsWith('## ')) {
        const title = line.trim().substring(3).trim();
        if (title) headings.push({ title, id: slug(title) });
      }
    });

    // ---- Reading time ----
    const readTime = readingTime(s);

    // ---- TOC ----
    let tocHtml = '';
    if (headings.length > 1) {
      tocHtml = '<div class="toc-sticky" id="toc"><h4>Contents</h4><ul>' +
        headings.map(h => '<li><a href="#' + h.id + '">' + escapeHtml(h.title) + '</a></li>').join('') +

      }

    const readingControls = `
      <div class="reading-controls">
        <button id="fontMinus" title="Decrease font size">A-</button>
        <button id="fontPlus" title="Increase font size">A+</button>
        <button id="lineHeightToggle" title="Toggle line spacing">Line spacing</button>
        <select id="fontFamilySelect" title="Choose font family">
          <option value="'PT Serif', Georgia, serif">PT Serif</option>

        </select>
      </div>
    `;

    container.innerHTML =
      '<div class="article-view">' +

      '<h1 class="headline">' + escapeHtml(s.headline) + '</h1>' +
      (s.dek ? '<p class="dek">' + escapeHtml(s.dek) + '</p>' : '') +
      '<span class="byline">' + (s.byline ? 'By ' + escapeHtml(s.byline) + ' · ' : '') + timeAgo(s.timestamp) + ' · 👁 ' + s.viewCount + ' · ' + readTime + (s.tags ? ' · #' + s.tags.split(',').map(t => t.trim()).join(' #') : '') + ' · <span class="save-star" data-id="' + s.id + '" style="cursor:pointer;">' + (isSaved(s.id) ? '⭐' : '☆') + '</span></span>' +
      '<img src="' + (s.imageUrl || placeholderFor(s.id)) + '" alt="' + escapeHtml(s.imageCaption || s.headline) + '" loading="lazy" onclick="window.openLightbox(\'' + s.id + '\', 0)">' +

      localStorage.setItem(FONTSIZE_KEY, String(size));
    });

      let size = parseFloat(bodyText.style.fontSize) || 18;
      size = Math.min(32, size + 2);
      bodyText.style.fontSize = size + 'px';
      localStorage.setItem(FONTSIZE_KEY, String(size));
    });
    lineHeightToggle.addEventListener('click', () => {
      let lh = parseFloat(bodyText.style.lineHeight) || 1.75;
      lh = lh === 1.75 ? 2.2 : 1.75;
      bodyText.style.lineHeight = lh;
      localStorage.setItem(LINEHEIGHT_KEY, String(lh));
    });

  async function handleDelete(id){
    if(trashTimer){ clearTimeout(trashTimer); trashTimer = null; }
    const s = stories.find(st => st.id === id);
    if(!s) return;
    if(!confirm('Delete "' + s.headline + '"?')) return;
    trashStory = s;
    await deleteStoryFromStorage(id);
    stories = stories.filter(st => st.id !== id);
    renderQueue();
    renderNav();
    populateArchiveFilter();
    const status = document.getElementById('saveStatus');
    if(status){ status.textContent = '🗑 Deleted. Undo? '; status.className = 'save-status';
      const undoBtn = document.createElement('button');
      undoBtn.textContent = 'Undo';
      undoBtn.className = 'btn-secondary btn btn-sm';
      undoBtn.onclick = async () => {
        if(trashStory){
          await saveStoryToStorage(trashStory);
          stories.unshift(trashStory);
          trashStory = null;
          if(trashTimer){ clearTimeout(trashTimer); trashTimer = null; }
          renderQueue(); renderNav(); populateArchiveFilter();
          if(status) status.textContent = 'Restored.';
        }
      };
      status.appendChild(undoBtn);
      trashTimer = setTimeout(() => { trashStory = null; if(status) status.textContent = ''; }, 30000);
    }
  }

  async function handleSubmit(e){
    e.preventDefault();
    const statusEl = document.getElementById('saveStatus');
    const publishBtn = document.getElementById('publishBtn');
    const editId = document.getElementById('editId').value;

    const headline = document.getElementById('headline').value.trim();
    const section = document.getElementById('section').value.trim();
    if(!headline || !section){ statusEl.textContent = 'Headline and section are required.'; statusEl.className = 'save-status error'; return; }

    const isBreaking = document.getElementById('isBreaking').checked;
    const story = {
      id: editId || ('story_' + Date.now() + '_' + Math.random().toString(36).slice(2,8)),
      headline, section,
      byline: document.getElementById('byline').value.trim(),
      dek: document.getElementById('dek').value.trim(),
      body: document.getElementById('body').value.trim(),
      imageUrl: document.getElementById('imageUrl').value.trim(),
      imageCaption: document.getElementById('imageCaption').value.trim(),
      tags: document.getElementById('tags').value.trim(),
      publishDate: document.getElementById('publishDate').value || null,
      isLead: document.getElementById('isLead').checked,
      isBrief: document.getElementById('isBrief').checked,
      isDraft: document.getElementById('isDraft').checked,
      isBreaking: isBreaking,
      timestamp: editId ? ((stories.find(s => s.id === editId) || {}).timestamp || Date.now()) : Date.now(),
      viewCount: editId ? ((stories.find(s => s.id === editId) || {}).viewCount || 0) : 0,
      comments: editId ? ((stories.find(s => s.id === editId) || {}).comments || []) : []
    };

    publishBtn.disabled = true;
    statusEl.textContent = 'Saving…';
    statusEl.className = 'save-status';

    try{
      if(story.isLead){
        for(const s of stories){
          if(s.id !== story.id && s.isLead){ s.isLead = false; await saveStoryToStorage(s); }
        }
      }
      await saveStoryToStorage(story);
      if(editId){ stories = stories.map(s => s.id === editId ? story : s); }
      else { stories.unshift(story); }
      statusEl.textContent = editId ? 'Updated.' : 'Published.';
      resetForm();
      renderNav();
      renderQueue();
      populateArchiveFilter();
      if(currentMode === 'front') renderFrontPage();

      if(isBreaking && !story.isDraft && (!story.publishDate || new Date(story.publishDate).getTime() <= Date.now())) {
        sendBreakingNotification(story.headline, story.dek);
      }
    }catch(err){ statusEl.textContent = 'Error saving.'; statusEl.className = 'save-status error'; }
    finally{ publishBtn.disabled = false; }
  }

  // -------- ARCHIVE FILTER --------
  function populateArchiveFilter(){
    const sel = document.getElementById('archiveFilter');
    if(!sel) return;
    const current = sel.value;
    const visible = getVisibleStories();
    const months = new Set();
    visible.forEach(s => {
      const d = new Date(s.timestamp);
      months.add(d.getFullYear() + '-' + String(d.getMonth()).padStart(2,'0'));
    });
    sel.innerHTML = '<option value="all">All time</option>';
    Array.from(months).sort().reverse().forEach(m => {
      const [year, month] = m.split('-').map(Number);
      const label = new Date(year, month, 1).toLocaleDateString('en-US', { month:'long', year:'numeric' });
      const opt = document.createElement('option');
      opt.value = m;
      opt.textContent = label;
      if(opt.value === current) opt.selected = true;
      sel.appendChild(opt);
    });
  }

  // -------- RSS FEED --------
  function renderRSS(){
    const items = getVisibleStories().slice(0, 20);
    const baseUrl = window.location.href.split('#')[0];
    let xml = '<?xml version="1.0" encoding="UTF-8"?>\n<rss version="2.0">\n<channel>\n';
    xml += '  <title>The Wheaton Weekly</title>\n';
    xml += '  <link>' + baseUrl + '</link>\n';
    xml += '  <description>All the news from Wheaton</description>\n';
    items.forEach(s => {
      xml += '  <item>\n';
      xml += '    <title>' + escapeXml(s.headline) + '</title>\n';
      xml += '    <link>' + baseUrl + '#/story/' + s.id + '</link>\n';
      xml += '    <description>' + escapeXml(s.dek || s.headline) + '</description>\n';
      xml += '    <pubDate>' + new Date(s.timestamp).toUTCString() + '</pubDate>\n';
      xml += '    <guid>' + baseUrl + '#/story/' + s.id + '</guid>\n';
      xml += '  </item>\n';
    });
    xml += '</channel>\n</rss>';
    document.body.innerHTML = '<pre style="font-family:monospace;white-space:pre-wrap;padding:20px;background:#f4f4f4;color:#000;max-width:900px;margin:40px auto;">' + escapeHtml(xml) + '</pre>';
  }

  // -------- KEYBOARD SHORTCUTS --------
  document.addEventListener('keydown', (e) => {
    if ((e.ctrlKey || e.metaKey) && e.key === 's') {
      e.preventDefault();
      if (currentMode === 'newsroom') {
        const form = document.getElementById('storyForm');
        if (form) form.dispatchEvent(new Event('submit'));
      }
    }
    if (e.key === 'Escape') {
      if (currentMode === 'article') goHome();
      if (currentMode === 'newsroom') {
        const cancelBtn = document.getElementById('cancelEditBtn');
        if (cancelBtn && cancelBtn.style.display !== 'none') resetForm();
      }
    }
  });

  }

  function showFront(){
    currentMode = 'front';
    location.hash = '';
    document.getElementById('frontPageView').style.display = '';
    document.getElementById('articleView').style.display = 'none';
    document.getElementById('newsroomView').style.display = 'none';

    document.getElementById('articleView').style.display = '';

      window.removeEventListener('scroll', window._progressHandler);

    renderArticle(id);
  }

  // -------- NEWSROOM GATE with reset button --------
  function showNewsroom(){
    currentMode = 'newsroom';
    document.getElementById('frontPageView').style.display = 'none';
    document.getElementById('articleView').style.display = 'none';
    document.getElementById('newsroomView').style.display = '';
    document.getElementById('sitePageView').style.display = 'none';
    setActiveButtons();
    renderNav();
    if(editorPassword && !checkSession()){
      const el = document.getElementById('newsroomView');
      el.innerHTML =
        '<div class="gate"><h2>Newsroom Access</h2><p>Enter the editor password.</p><div class="gate-error" id="gateError"></div>' +
        '<input type="password" id="gatePassword" placeholder="Password" autofocus>' +
        '<div class="gate-actions">' +
          '<button class="btn btn-publish" id="gateSubmit">Unlock</button>' +
          '<button class="btn btn-secondary" id="gateReset">Reset Password</button>' +
        '</div>' +
        '<div class="gate-note">Session expires after 30 min of inactivity. Click "Reset Password" to remove the password and unlock the Newsroom.</div></div>';
      const submit = async () => {
        const val = document.getElementById('gatePassword').value;
        const hashed = await hashPassword(val);
        if(hashed === editorPassword){ isAuthed = true; touchSession(); showNewsroom(); }
        else { document.getElementById('gateError').textContent = 'Incorrect password.'; }
      };
      document.getElementById('gateSubmit').onclick = submit;
      document.getElementById('gatePassword').addEventListener('keydown', ev => { if(ev.key === 'Enter') submit(); });
      document.getElementById('gateReset').onclick = async () => {
        if(confirm('Remove the password and unlock the Newsroom? (You can set a new one later.)')){
          await clearPassword();
          isAuthed = true;
          showNewsroom();
        }
      };
      return;
    }
    renderNewsroomShell();
    restoreDraft();
    if (window._progressHandler) {
      window.removeEventListener('scroll', window._progressHandler);
      window._progressHandler = null;
    }
    document.getElementById('progressBar').style.width = '0%';
  }

  function handleHashChange(){
    const hash = location.hash;
    if (hash === '#/feed') {
      renderRSS();
      return;
    }
    const pageMatch = hash.match(/^#\/page\/([a-z0-9-]+)$/i);
    if(pageMatch){ showSitePage(decodeURIComponent(pageMatch[1]).toLowerCase()); return; }
    const pm = hash.match(/^#\/page\/(.+)$/);
    if(pm){ showSitePage(decodeURIComponent(pm[1])); return; }
    const m = hash.match(/^#\/story\/(.+)$/);
    if(m){ showArticle(decodeURIComponent(m[1])); }
    else if(currentMode !== 'newsroom'){ showFront(); }
  }

  async function init(){
    loadTheme();
    const savedFont = localStorage.getItem(FONT_KEY);
    if (savedFont) {
      document.documentElement.style.setProperty('--body-font', savedFont);
      document.getElementById('fontSelector').value = savedFont;
    }
    document.getElementById('frontPageView').innerHTML = '<div class="loading-note">Loading stories…</div>';
    await Promise.all([loadStories(), loadPassword(), loadSitePages()]);
    recordAnalytics('visits');
    if(editorPassword) { if(checkSession()) touchSession(); else isAuthed = false; }
    populateArchiveFilter();
    const pageMatch = location.hash.match(/^#\/page\/([a-z0-9-]+)$/i);
    const m = location.hash.match(/^#\/story\/(.+)$/);
    if(pageMatch){ showSitePage(decodeURIComponent(pageMatch[1]).toLowerCase()); }
    else if(m){ showArticle(decodeURIComponent(m[1])); }
    else { showFront(); }
    updateSavedUI();
  }
  init();
})();
